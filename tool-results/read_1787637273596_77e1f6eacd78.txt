    18→// Animated counter hook
    19→function useCounter(end: number, duration: number = 2000) {
    20→  const [count, setCount] = useState(0)
    21→  
    22→  useEffect(() => {
    23→    let start = 0
    24→    const increment = end / (duration / 16)
    25→    const timer = setInterval(() => {
    26→      start += increment
    27→      if (start >= end) {
    28→        setCount(end)
    29→        clearInterval(timer)
    30→      } else {
    31→        setCount(Math.floor(start))
    32→      }
    33→    }, 16)
    34→    return () => clearInterval(timer)
    35→  }, [end, duration])
    36→  
    37→  return count
    38→}
